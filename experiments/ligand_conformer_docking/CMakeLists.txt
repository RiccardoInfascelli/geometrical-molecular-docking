cmake_minimum_required(VERSION 3.20)
project(ligand_conformer_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 REQUIRED)

option(ENABLE_CUDA "Build CUDA conformer docking (OFF supports generation and host tests)" ON)
set(BUILD_EXPERIMENTS OFF CACHE BOOL "Build other engine experiments")
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)

add_library(conformer_generation conformer_generation.cpp)
target_include_directories(conformer_generation PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(conformer_generation PUBLIC docking_gpu_output
    RDKitDistGeomHelpers RDKitForceFieldHelpers RDKitForceField RDKitMolAlign RDKitRDGeometryLib)
add_library(conformer_quality_support conformer_quality_input.cpp quaternion_rmsd.cpp)
target_include_directories(conformer_quality_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(conformer_quality_support PUBLIC conformer_generation Eigen3::Eigen)
add_executable(conformer_quality conformer_quality.cpp)
target_link_libraries(conformer_quality PRIVATE conformer_quality_support)
add_library(ensemble_ranking ensemble_ranking.cpp)
target_include_directories(ensemble_ranking PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ensemble_ranking PUBLIC docking_rescoring)
add_library(ensemble_diagnostics ensemble_diagnostics.cpp)
target_include_directories(ensemble_diagnostics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ensemble_diagnostics PUBLIC docking_gpu_output)
add_executable(ligand_conformer_docking ligand_conformer_docking.cpp)
target_link_libraries(ligand_conformer_docking PRIVATE conformer_generation ensemble_ranking ensemble_diagnostics)
if(ENABLE_CUDA)
    add_library(conformer_cuda_search conformer_docking.cpp)
    target_include_directories(conformer_cuda_search PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(conformer_cuda_search PUBLIC ensemble_ranking docking_gpu_search docking_gpu_output docking_cuda_rescoring)
    target_link_libraries(ligand_conformer_docking PRIVATE conformer_cuda_search)
    target_compile_definitions(ligand_conformer_docking PRIVATE CONFORMER_WITH_CUDA)
endif()

enable_testing()
add_executable(conformer_quality_smoke conformer_quality_smoke.cpp)
target_link_libraries(conformer_quality_smoke PRIVATE conformer_quality_support)
add_test(NAME conformer_quality_smoke COMMAND conformer_quality_smoke)
add_executable(conformer_generation_test conformer_generation_test.cpp)
target_link_libraries(conformer_generation_test PRIVATE conformer_generation)
add_test(NAME conformer_generation COMMAND conformer_generation_test)
add_executable(ensemble_ranking_test ensemble_ranking_test.cpp)
target_link_libraries(ensemble_ranking_test PRIVATE ensemble_ranking)
add_test(NAME ensemble_ranking COMMAND ensemble_ranking_test)
add_executable(ensemble_diagnostics_test ensemble_diagnostics_test.cpp)
target_link_libraries(ensemble_diagnostics_test PRIVATE ensemble_diagnostics)
add_test(NAME ensemble_diagnostics COMMAND ensemble_diagnostics_test)
add_executable(rmsd_oracle_test rmsd_oracle_test.cpp)
target_link_libraries(rmsd_oracle_test PRIVATE ensemble_ranking docking_gpu_output)
add_test(NAME rmsd_oracle COMMAND rmsd_oracle_test)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME conformer_single_run_workflow COMMAND ${Python3_EXECUTABLE} -B
        ${CMAKE_CURRENT_SOURCE_DIR}/test_run_1a1e.py)
    add_test(NAME conformer_quality_runner COMMAND ${Python3_EXECUTABLE} -B
        ${CMAKE_CURRENT_SOURCE_DIR}/test_conformer_quality_runner.py)
    set_tests_properties(conformer_quality_runner PROPERTIES
        ENVIRONMENT "CONFORMER_QUALITY_EXECUTABLE=$<TARGET_FILE:conformer_quality>"
        LABELS "cpu;synthetic;python" TIMEOUT 30)
endif()
if(ENABLE_CUDA AND NOT DOCKING_TEST_COMPLEX STREQUAL "")
    add_executable(conformer_original_test conformer_original_test.cpp)
    target_link_libraries(conformer_original_test PRIVATE conformer_cuda_search docking_gpu_screen)
    add_test(NAME conformer_original COMMAND conformer_original_test "${DOCKING_TEST_COMPLEX}")
    add_executable(conformer_execution_test conformer_execution_test.cpp)
    target_link_libraries(conformer_execution_test PRIVATE conformer_cuda_search)
    add_test(NAME conformer_execution COMMAND conformer_execution_test "${DOCKING_TEST_COMPLEX}")
    set_tests_properties(conformer_execution PROPERTIES LABELS "cuda;integration" TIMEOUT 1800)
endif()

add_executable(conformer_export_test conformer_export_test.cpp)
target_link_libraries(conformer_export_test PRIVATE conformer_generation ensemble_ranking ensemble_diagnostics docking_gpu_output)
add_test(NAME conformer_export COMMAND conformer_export_test)
